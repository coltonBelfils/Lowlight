// std/math — Mathematical constants, rounding, powers, logarithms, and trigonometry.
//
// Implemented in pure Lowlight over 64-bit IEEE-754 floats (`float`) and integers (`int`).

// Ratio of a circle's circumference to its diameter (`pi`).
bind PI: float = 3.14159265358979323846

// Full turn circle constant (`2 * pi`).
bind TAU: float = 6.28318530717958647692

// Euler's number (`e`), base of natural logarithms.
bind E: float = 2.71828182845904523536

// Natural logarithm of `2`.
bind LN2: float = 0.69314718055994530942

// Natural logarithm of `10`.
bind LN10: float = 2.30258509299404568402

// Square root of `2`.
bind SQRT2: float = 1.41421356237309504880

// Square root of `3`.
bind SQRT3: float = 1.73205080756887729353

// Positive IEEE-754 infinity (`+Inf`).
bind INF: float = 1.0e308 * 10.0

// Quiet IEEE-754 Not-a-Number (`NaN`).
bind NAN: float = INF - INF

// Reports whether `x` is IEEE-754 Not-a-Number (`NaN`).
bind is_nan = (x: float) bool: not (x == x)

// Reports whether `x` is positive or negative IEEE-754 infinity (`+Inf` or `-Inf`).
bind is_inf = (x: float) bool: x == INF or x == -INF

// Returns the absolute value (`|x|`) of floating-point number `x`.
bind abs_float = (x: float) float:
	if(x < 0.0): -x else: x

// Returns the absolute value of `x` (alias for [`abs_float`]).
bind fabs = (x: float) float: abs_float(x)

// Truncates `x` toward zero, discarding any fractional digits.
bind trunc = (x: float) float: {
	if(is_nan(x) or abs_float(x) >= 4503599627370496.0): func <- x
	float(int(x))
}

// Returns the largest integer value (as [`float`]) less than or equal to `x`.
bind floor = (x: float) float: {
	let t = trunc(x)
	if(x < t): t - 1.0 else: t
}

// Returns the smallest integer value (as [`float`]) greater than or equal to `x`.
bind ceil = (x: float) float: {
	let t = trunc(x)
	if(x > t): t + 1.0 else: t
}

// Rounds `x` to the nearest integer (as [`float`]), rounding halves away from zero.
//
// Exact for every input: the fractional part is computed as `x - trunc(x)` (exact in IEEE-754)
// rather than by adding `0.5`, which misrounds `0.49999999999999994` and large odd integers.
// Values of magnitude `2^52` or more are already integers and are returned unchanged.
bind round = (x: float) float: {
	if(is_nan(x) or is_inf(x) or abs_float(x) >= 4503599627370496.0): func <- x
	let t = trunc(x)
	let d = x - t
	let r = if(d >= 0.5): t + 1.0 else if(d <= -0.5): t - 1.0 else: t
	if(r == 0.0 and x < 0.0): -0.0 else: r
}

// Returns the IEEE-754 floating-point remainder of `x / y` (`NaN` if `x` is `±Inf`, `y == 0.0`, or either is `NaN`; `x` if `y` is `±Inf` and `x` is finite).
bind fmod = (x: float, y: float) float: {
	if(is_nan(x) or is_nan(y) or is_inf(x) or y == 0.0): func <- NAN
	if(is_inf(y) or x == 0.0): func <- x
	let ax = abs_float(x)
	let ay = abs_float(y)
	if(ax < ay): func <- x
	if(ax == ay): {
		if(x < 0.0): func <- -0.0
		func <- 0.0
	}
	let p64 = 18446744073709551616.0
	let pm64 = 1.0 / p64
	let d = ay
	let k = 0
	for(d <= ax * pm64): {
		d *= p64
		k += 64
	}
	for(d <= ax * 0.5): {
		d *= 2.0
		k += 1
	}
	for(ax >= ay): {
		if(ax >= d): ax -= d
		if(ax >= ay): {
			for(k >= 64 and ax < d * pm64): {
				d *= pm64
				k -= 64
			}
			for(k > 0 and ax < d): {
				d *= 0.5
				k -= 1
			}
		}
	}
	if(x < 0.0): -ax else: ax
}

// Computes the non-negative square root of `x` (`NaN` for `x < 0.0` or `NaN`, `+Inf` for `+Inf`, `±0.0` preserved).
bind sqrt = (x: float) float: {
	if(is_nan(x) or x < 0.0): func <- NAN
	if(x == 0.0 or x == INF): func <- x
	let scale = 1.0
	let m = x
	let p64 = 18446744073709551616.0
	let pm64 = 1.0 / p64
	let s32 = 4294967296.0
	let sm32 = 1.0 / s32
	for(m >= p64): {
		m *= pm64
		scale *= s32
	}
	for(m < pm64): {
		m *= p64
		scale *= sm32
	}
	for(m >= 4.0): {
		m *= 0.25
		scale *= 2.0
	}
	for(m < 0.25): {
		m *= 4.0
		scale *= 0.5
	}
	let g = 0.5 * (1.0 + m)
	let i = 0
	for(i < 8): {
		g = 0.5 * (g + m / g)
		i += 1
	}
	g * scale
}

// Computes the real cube root (`x^(1/3)`) of `x`, preserving its sign.
//
// `cbrt(±0)`, `cbrt(±inf)`, and `cbrt(nan)` return their argument. The argument is first scaled
// by powers of 8 into `[1, 8)` (exactly, since `cbrt(8^k · m) = 2^k · cbrt(m)`), so the Newton
// iteration converges for every magnitude, including subnormals.
bind cbrt = (x: float) float: {
	if(is_nan(x) or is_inf(x) or x == 0.0): func <- x
	let is_neg = x < 0.0
	let m = if(is_neg): -x else: x
	let scale = 1.0
	let big = 1152921504606846976.0      // 2^60 = 8^20
	let big_root = 1048576.0             // 2^20
	while(m >= big): {
		m /= big
		scale *= big_root
	}
	while(m < 1.0): {
		m *= big
		scale /= big_root
	}
	while(m >= 8.0): {
		m /= 8.0
		scale *= 2.0
	}
	// Newton's method from above converges monotonically to cbrt(m) in [1, 2).
	let g = 2.0
	for(i in 0..8): g = (2.0 * g + m / (g * g)) / 3.0
	let r = g * scale
	if(is_neg): -r else: r
}

// Returns the Euclidean length `sqrt(x * x + y * y)` without overflow or underflow in the
// intermediate squares: `hypot(1e200, 1e200)` is `1.414e200`. Infinite arguments give `inf`
// (even if the other is `nan`); otherwise a `nan` argument gives `nan`.
bind hypot = (x: float, y: float) float: {
	if(is_inf(x) or is_inf(y)): func <- INF
	if(is_nan(x) or is_nan(y)): func <- NAN
	let ax = abs_float(x)
	let ay = abs_float(y)
	let big = if(ax > ay): ax else: ay
	let small = if(ax > ay): ay else: ax
	if(big == 0.0): func <- 0.0
	let q = small / big
	big * sqrt(1.0 + q * q)
}

bind _scale_pow2 = (v: float, k: int) float: {
	let res = v
	let rem = k
	if(rem > 0): {
		for(rem > 0): {
			let step = if(rem > 512): 512 else: rem
			let base = 2.0
			let e = step
			for(e > 0): {
				if((e & 1) != 0): res *= base
				e >>= 1
				if(e > 0): base *= base
			}
			rem -= step
		}
	} else if(rem < 0): {
		let neg_rem = -rem
		for(neg_rem > 0): {
			let step = if(neg_rem > 512): 512 else: neg_rem
			let base = 0.5
			let e = step
			for(e > 0): {
				if((e & 1) != 0): res *= base
				e >>= 1
				if(e > 0): base *= base
			}
			neg_rem -= step
		}
	}
	res
}

bind _exp_scaled = (x: float, extra_k: int) float: {
	let ln2_hi = 0.693147180369123816490
	let ln2_lo = 1.90821492927058770002e-10
	let k = int(round(x / LN2))
	let r = (x - float(k) * ln2_hi) - float(k) * ln2_lo
	let term = 1.0
	let sum = 1.0
	let n = 1
	for(n <= 16): {
		term = term * r / float(n)
		sum += term
		n += 1
	}
	_scale_pow2(sum, k + extra_k)
}

// Computes the base-[`E`] exponential function (`e^x`), handling `NaN`, `±Inf`, overflow (`> 710.0`), and subnormal/underflow (`< -745.2`).
bind exp = (x: float) float: {
	if(is_nan(x)): func <- NAN
	if(x == 0.0): func <- 1.0
	if(x > 710.0): func <- INF
	if(x < -745.2): func <- 0.0
	_exp_scaled(x, 0)
}

// Computes the base-2 exponential function (`2^x`), handling `NaN`, `±Inf`, overflow (`>= 1024.0`), and subnormal/underflow (`<= -1075.0`).
bind exp2 = (x: float) float: {
	if(is_nan(x)): func <- NAN
	if(x == 0.0): func <- 1.0
	if(x >= 1024.0): func <- INF
	if(x <= -1075.0): func <- 0.0
	let k = int(round(x))
	let f = x - float(k)
	let sum = 1.0
	if(f != 0.0): {
		let r = f * LN2
		let term = 1.0
		let n = 1
		for(n <= 16): {
			term = term * r / float(n)
			sum += term
			n += 1
		}
	}
	_scale_pow2(sum, k)
}

// Computes the natural logarithm `ln(x)` (`NaN` for `x < 0.0` or `NaN`, `-Inf` for `±0.0`, `+Inf` for `+Inf`).
bind log = (x: float) float: {
	if(is_nan(x) or x < 0.0): func <- NAN
	if(x == 0.0): func <- -INF
	if(x == INF): func <- INF
	if(x == 1.0): func <- 0.0
	let k = 0
	let m = x
	let p64 = 18446744073709551616.0
	let pm64 = 1.0 / p64
	for(m >= p64): {
		m *= pm64
		k += 64
	}
	for(m < pm64): {
		m *= p64
		k -= 64
	}
	for(m >= SQRT2): {
		m *= 0.5
		k += 1
	}
	for(m < 0.70710678118654752440): {
		m *= 2.0
		k -= 1
	}
	let u = (m - 1.0) / (m + 1.0)
	let u2 = u * u
	let term = u
	let sum = u
	let n = 1
	for(n <= 15): {
		term *= u2
		sum += term / float(2 * n + 1)
		n += 1
	}
	2.0 * sum + float(k) * LN2
}

// Computes the base-10 logarithm of `x`.
bind log10 = (x: float) float: log(x) / LN10

// Computes the base-2 logarithm of `x` (`NaN` for `x < 0.0` or `NaN`, `-Inf` for `±0.0`, `+Inf` for `+Inf`).
bind log2 = (x: float) float: {
	if(is_nan(x) or x < 0.0): func <- NAN
	if(x == 0.0): func <- -INF
	if(x == INF): func <- INF
	if(x == 1.0): func <- 0.0
	let k = 0
	let m = x
	let p64 = 18446744073709551616.0
	let pm64 = 1.0 / p64
	for(m >= p64): {
		m *= pm64
		k += 64
	}
	for(m < pm64): {
		m *= p64
		k -= 64
	}
	for(m >= SQRT2): {
		m *= 0.5
		k += 1
	}
	for(m < 0.70710678118654752440): {
		m *= 2.0
		k -= 1
	}
	let u = (m - 1.0) / (m + 1.0)
	let u2 = u * u
	let term = u
	let sum = u
	let n = 1
	for(n <= 15): {
		term *= u2
		sum += term / float(2 * n + 1)
		n += 1
	}

}

bind _is_odd_int = (y: float) bool: {
	if(is_nan(y) or is_inf(y)): func <- false
	if(y != trunc(y)): func <- false
	let ay = abs_float(y)
	if(ay >= 9007199254740992.0): func <- false
	(int(ay) & 1) != 0
}

// Computes `base` raised to `exponent` (`base^exponent`), following IEEE-754 special-case rules.
bind pow = (base: float, exponent: float) float: {
	if(exponent == 0.0): func <- 1.0
	if(base == 1.0): func <- 1.0
	if(is_nan(base) or is_nan(exponent)): func <- NAN
	if(base == 0.0): {
		let is_neg_zero = (1.0 / base) < 0.0
		if(exponent < 0.0): {
			if(is_neg_zero and _is_odd_int(exponent)): func <- -INF
			func <- INF
		}
		if(is_neg_zero and _is_odd_int(exponent)): func <- -0.0
		func <- 0.0
	}
	if(base == -1.0 and is_inf(exponent)): func <- 1.0
	if(is_inf(exponent)): {
		let ab = abs_float(base)
		if(exponent == INF): {
			if(ab > 1.0): func <- INF
			func <- 0.0
		}
		if(ab > 1.0): func <- 0.0
		func <- INF
	}
	if(is_inf(base)): {
		if(base == INF): {
			if(exponent > 0.0): func <- INF
			func <- 0.0
		}
		if(exponent > 0.0): {
			if(_is_odd_int(exponent)): func <- -INF
			func <- INF
		}
		if(_is_odd_int(exponent)): func <- -0.0
		func <- 0.0
	}
	if(base < 0.0): {
		if(exponent != trunc(exponent)): func <- NAN
		if(base == -1.0): {
			if(_is_odd_int(exponent)): func <- -1.0
			func <- 1.0
		}
		let mag = pow(-base, exponent)
		if(_is_odd_int(exponent)): func <- -mag
		func <- mag
	}
	let exp_trunc = trunc(exponent)
	if(exponent == exp_trunc and abs_float(exponent) <= 1048576.0): {
		let n = int(exp_trunc)
		let is_neg_exp = n < 0
		let p = if(is_neg_exp): -n else: n
		let acc = 1.0
		let b = base
		for(p > 0): {
			if((p & 1) == 1): acc *= b
			p = p >> 1
			if(p > 0): b *= b
		}
		if(is_neg_exp): {
			if(not is_inf(acc)): func <- 1.0 / acc
		} else: {
			func <- acc
		}
	}
	exp(exponent * log(base))
}

bind _PIO2_HI = 1.5707963267948966
bind _PIO2_LO = 6.123233995736766e-17

bind _float_bits_of = (v: float) u64:
	unsafe {
		let vp = addr_of(v)
		*ptr_cast[u64](vp)
	}

// Returns the 32 bits of 2/π starting at bit `pos` (bit 0 is the first after the point).
bind _two_over_pi_bits = (pos: int) u64: {
	let w = pos / 32
	let b = pos % 32
	let hi = _TWO_OVER_PI.get(w) ?? u64(0)
	if(b == 0): func <- hi
	let lo = _TWO_OVER_PI.get(w + 1) ?? u64(0)
	((hi << u64(b)) | (lo >> u64(32 - b))) & u64(0xffffffff)
}

// Returns `n <= 32` bits of a little-endian 32-bit-limb integer, starting at bit `pos`.
bind _limb_bits = (limbs: [u64], pos: int, n: int) u64: {
	let w = pos / 32
	let b = pos % 32
	let lo = limbs.get(w) ?? u64(0)
	let hi = limbs.get(w + 1) ?? u64(0)
	let joined = (lo >> u64(b)) | (if(b == 0): u64(0) else: hi << u64(32 - b))
	let mask = (u64(1) << u64(n)) - u64(1)
	joined & mask
}

// Payne–Hanek argument reduction: returns `+[q, r]` with `x = (4k + q)·π/2 + r`, `q` in `0..4`,
// and `r` in `[-π/4, π/4]`, computed exactly from the binary expansion of 2/π. Unlike
// `fmod(x, TAU)`, whose error grows with `x` because `TAU` is rounded, this is accurate for every
// finite double (`sin(1e15)`, `cos(1e300)`).
bind _rem_pio2 = (x: float) +[int, float]: {
	let bits = _float_bits_of(abs_float(x))
	let exp_field = int((bits >> u64(52)) & u64(0x7ff))
	let frac = bits & u64(0xfffffffffffff)
	let mant = if(exp_field == 0): frac else: frac | (u64(1) << u64(52))
	let e = if(exp_field == 0): -1074 else: exp_field - 1075      // |x| = mant · 2^e

	// Only bits of 2/π from position e - 2 onward affect (x · 2/π) mod 4.
	let j0 = if(e - 2 > 0): e - 2 else: 0
	let s = e - j0
	// 192 bits of 2/π as six 32-bit limbs, least significant first.
	let w = for(k in 0..6) [u64]: _two_over_pi_bits(j0 + 32 * (5 - k))
	let m = [mant & u64(0xffffffff), mant >> u64(32)]

	// P = mant · W, eight 32-bit limbs, least significant first.
	let p: [u64] = [u64(0), u64(0), u64(0), u64(0), u64(0), u64(0), u64(0), u64(0)]
	for(i in 0..6): {
		let carry = u64(0)
		for(j in 0..2): {
			let t = w[i] * m[j] + p[i + j] + carry
			p[i + j] = t & u64(0xffffffff)
			carry = t >> u64(32)
		}
		p[i + 2] = carry
	}

	// x · 2/π mod 4 = P · 2^(s - 192): the quadrant is the two bits above position 192 - s,
	// and the fraction is the bits below it.
	let point = 192 - s
	let q = int(_limb_bits(p, point, 2))
	// The 128 fraction bits below the point, most significant chunk first.
	let chunks = for(k in 1..=4) [u64]: _limb_bits(p, point - 32 * k, 32)
	// A fraction of one half or more is taken as (fraction - 1) in the next quadrant. Negate it
	// exactly in 128-bit two's complement before converting, so values just below a multiple of
	// π/2 keep their precision.
	let negative = chunks[0] >= u64(0x80000000)
	if(negative): {
		q += 1
		let carry = u64(1)
		for(i in [3, 2, 1, 0]): {
			let t = (chunks[i] ^ u64(0xffffffff)) + carry
			chunks[i] = t & u64(0xffffffff)
			carry = t >> u64(32)
		}
	}
	let f = 0.0
	for(i in [3, 2, 1, 0]): f = (f + float(chunks[i])) / 4294967296.0
	if(negative): f = -f
	let r = f * _PIO2_HI + f * _PIO2_LO
	if(x < 0.0): +[(4 - q % 4) % 4, -r] else: +[q % 4, r]
}

bind _reduce_angle = (x: float) float: {
	if(is_nan(x) or is_inf(x)): func <- NAN
	let r = fmod(x, TAU)
	if(r > PI): r -= TAU
	else if(r < -PI): r += TAU
	r
}

bind _sin_poly = (r: float) float: {
	let r2 = r * r
	let term = r
	let sum = r
	let n = 1
	for(n <= 10): {
		term = -term * r2 / float(2 * n * (2 * n + 1))
		sum += term
		n += 1
	}
	sum
}

bind _cos_poly = (r: float) float: {
	let r2 = r * r
	let term = 1.0
	let sum = 1.0
	let n = 1
	for(n <= 10): {
		term = -term * r2 / float((2 * n - 1) * 2 * n)
		sum += term
		n += 1
	}
	sum
}

// Computes the sine of angle `x` (in radians), returning `NaN` for `NaN` and `±Inf`.
bind sin = (x: float) float: {
	if(is_nan(x) or is_inf(x)): func <- NAN
	if(abs_float(x) <= 0.7853981633974483): func <- _sin_poly(x)
	let q, r = unpack _rem_pio2(x)
	match(q):
		when 0: _sin_poly(r)
		when 1: _cos_poly(r)
		when 2: -_sin_poly(r)
		default: -_cos_poly(r)
}

// Computes the cosine of angle `x` (in radians), returning `NaN` for `NaN` and `±Inf`.
bind cos = (x: float) float: {
	if(is_nan(x) or is_inf(x)): func <- NAN
	if(abs_float(x) <= 0.7853981633974483): func <- _cos_poly(x)
	let q, r = unpack _rem_pio2(x)
	match(q):
		when 0: _cos_poly(r)
		when 1: -_sin_poly(r)
		when 2: -_cos_poly(r)
		default: _sin_poly(r)
}

// Computes the tangent (`sin(x) / cos(x)`) of angle `x` (in radians).
bind tan = (x: float) float: {
	if(is_nan(x) or is_inf(x)): func <- NAN
	if(abs_float(x) <= 0.7853981633974483): func <- _sin_poly(x) / _cos_poly(x)
	let q, r = unpack _rem_pio2(x)
	if(q % 2 == 0): _sin_poly(r) / _cos_poly(r) else: -_cos_poly(r) / _sin_poly(r)
}

bind _atan_unit = (x: float) float: {
	let offset = 0.0
	let t = x
	if(t > 2.0 - SQRT3): {
		offset = PI / 6.0
		t = (SQRT3 * t - 1.0) / (SQRT3 + t)
	}
	let t2 = t * t
	let term = t
	let sum = t
	let n = 1
	for(n <= 12): {
		term = -term * t2
		sum += term / float(2 * n + 1)
		n += 1
	}
	offset + sum
}

// Computes the principal value arctangent of `x` in radians (`-PI/2..=PI/2`).
bind atan = (x: float) float: {
	if(x == 0.0): func <- 0.0
	let is_neg = x < 0.0
	let ax = if(is_neg): -x else: x
	let res = if(ax > 1.0): PI * 0.5 - _atan_unit(1.0 / ax) else: _atan_unit(ax)
	if(is_neg): -res else: res
}

// Computes the four-quadrant inverse tangent of `y / x` in radians (`-PI..=PI`), returning `0.0` at `(0, 0)`.
bind atan2 = (y: float, x: float) float: {
	if(x > 0.0): func <- atan(y / x)
	if(x < 0.0): {
		if(y >= 0.0): func <- atan(y / x) + PI
		func <- atan(y / x) - PI
	}
	if(y > 0.0): func <- PI * 0.5
	if(y < 0.0): func <- -PI * 0.5
	0.0
}

// Computes the inverse sine (`arcsin(x)`) in radians, clamping inputs outside `-1.0..=1.0`.
bind asin = (x: float) float: {
	if(x >= 1.0): func <- PI * 0.5
	if(x <= -1.0): func <- -PI * 0.5
	atan2(x, sqrt(1.0 - x * x))
}

// Computes the inverse cosine (`arccos(x)`) in radians, clamping inputs outside `-1.0..=1.0`.
bind acos = (x: float) float: {
	if(x >= 1.0): func <- 0.0
	if(x <= -1.0): func <- PI
	atan2(sqrt(1.0 - x * x), x)
}

// Computes the hyperbolic sine of `x`, handling `NaN`, `±Inf`, small-angle precision, and large values up to `~710.475`.
bind sinh = (x: float) float: {
	if(is_nan(x)): func <- NAN
	if(is_inf(x) or x == 0.0): func <- x
	let ax = abs_float(x)
	if(ax < 1.0): {
		let x2 = x * x
		let term = x
		let sum = x
		let n = 1
		for(n <= 10): {
			term = term * x2 / float(2 * n * (2 * n + 1))
			sum += term
			n += 1
		}
		func <- sum
	}
	if(ax <= 22.0): {
		let e = exp(ax)
		let res = 0.5 * (e - 1.0 / e)
		if(x < 0.0): func <- -res
		func <- res
	}
	if(ax <= 711.0): {
		let res = _exp_scaled(ax, -1)
		if(x < 0.0): func <- -res
		func <- res
	}
	if(x < 0.0): -INF else: INF
}

// Computes the hyperbolic cosine of `x`, handling `NaN`, `±Inf`, and large values up to `~710.475`.
bind cosh = (x: float) float: {
	if(is_nan(x)): func <- NAN
	if(is_inf(x)): func <- INF
	if(x == 0.0): func <- 1.0
	let ax = abs_float(x)
	if(ax <= 22.0): {
		let e = exp(ax)
		func <- 0.5 * (e + 1.0 / e)
	}
	if(ax <= 711.0): func <- _exp_scaled(ax, -1)
	INF
}

// Computes the hyperbolic tangent of `x` (`±1.0` for `±Inf` and `|x| >= 20.0`, `NaN` for `NaN`).
bind tanh = (x: float) float: {
	if(is_nan(x)): func <- NAN
	if(x == 0.0): func <- x
	let ax = abs_float(x)
	if(ax >= 20.0): {
		if(x < 0.0): func <- -1.0
		func <- 1.0
	}
	if(ax < 1.0): func <- sinh(x) / cosh(x)
	let e2 = exp(2.0 * ax)
	let res = (e2 - 1.0) / (e2 + 1.0)
	if(x < 0.0): -res else: res
}

bind clamp_int = (val: int, lo: int, hi: int) int:
	if(val < lo): lo
	else if(val > hi): hi
	else: val

@test("math round, cbrt, and hypot are exact or overflow-safe at the extremes")
bind test_math_extremes = (): {
	assert(round(0.49999999999999994) == 0.0, "largest double below 0.5 rounds down")
