// std/unicode — Unicode-aware case conversion and case-insensitive comparison.
//
// The prelude's `string.to_upper` and `string.to_lower` change only ASCII letters, which is
// fast and exactly right for protocols, identifiers, and file formats. This package implements
// the full Unicode default case operations for human-language text:
//
// - [`to_upper`] and [`to_lower`] apply the complete mappings, including those that change
//   length (`"ß"` uppercases to `"SS"`, `"ﬁ"` to `"FI"`, `"İ"` lowercases to `"i̇"`) and the
//   Final_Sigma rule (`"ΟΔΟΣ"` lowercases to `"οδος"`, with a final `ς`).
// - [`fold`] applies full case folding, the form to compare text case-insensitively, and
//   [`equal_fold`] compares with it.
//
// The mappings are language-independent: there are no Turkish/Azeri dotted-i or Lithuanian
// rules. Tables follow the Unicode version in [`UNICODE_VERSION`] and are generated by
// `std/unicode/gen_tables.py`.
//
// ```lowlight
// import "std/unicode"
//
// let shout = unicode.to_upper("straße")              // "STRASSE"
// let same = unicode.equal_fold("Straße", "STRASSE")   // true
// ```

// Returns `text` with every character converted to uppercase.
//
// Uses full mappings, so the result may be longer than `text` (`"ß"` becomes `"SS"`).
const to_upper = fn(text: string) string: _convert(text, 0)

// Returns `text` with every character converted to lowercase.
//
// Uses full mappings and applies the Final_Sigma rule: a capital sigma `Σ` at the end of a
// word lowercases to `ς`, and elsewhere to `σ`.
const to_lower = fn(text: string) string: _convert(text, 1)

// Returns the full case folding of `text`: a caseless form for comparing text
// case-insensitively (`"Straße"`, `"STRASSE"`, and `"strasse"` all fold to `"strasse"`).
// Folded text is meant for comparison and lookup, not for display.
const fold = fn(text: string) string: _convert(text, 2)

// Reports whether `a` and `b` are equal ignoring case, by comparing their full case foldings.
const equal_fold = fn(a: string, b: string) bool: fold(a) == fold(b)

// Converts `text` with mode 0 (upper), 1 (lower), or 2 (fold).
const _convert = fn(text: string, mode: int) string: {
	let out = string.with_capacity(text.len())
	let cs = text.chars()
	for(i, c in cs): {
		let cp = int(c)
		if(cp < 128): {
			// ASCII fast path: only letters change, one to one.
			if(mode == 0 and cp >= 97 and cp <= 122): out.push_byte(u8(cp - 32))
			else if(mode != 0 and cp >= 65 and cp <= 90): out.push_byte(u8(cp + 32))
			else: out.push_byte(u8(cp))
			next <-
		}
		if(mode == 1 and cp == 0x3A3): {
			out.push_char(if(_is_final_sigma(cs, i)): 'ς' else: 'σ')
			next <-
		}
		let k = match(mode):
			when 0: _special_index(_UPPER_SPECIAL_KEYS, cp)
			when 1: _special_index(_LOWER_SPECIAL_KEYS, cp)
			default: _special_index(_FOLD_SPECIAL_KEYS, cp)
		if(k >= 0): {
			match(mode):
				when 0: out.push(_UPPER_SPECIAL_VALUES[k])
				when 1: out.push(_LOWER_SPECIAL_VALUES[k])
				default: out.push(_FOLD_SPECIAL_VALUES[k])
			next <-
		}
		let mapped = match(mode):
			when 0: _map_simple(_UPPER_RANGES, cp)
			when 1: _map_simple(_LOWER_RANGES, cp)
			default: _map_simple(_FOLD_RANGES, cp)
		out.push_char(char(mapped))
	}
	out
}

// Maps `cp` through a table of (start, end, delta, stride) runs, or returns it unchanged.
const _map_simple = fn(table: [int], cp: int) int: {
	let lo = 0
	let hi = table.len() / 4 - 1
	while(lo <= hi): {
		let mid = (lo + hi) / 2
		let start = table[mid * 4]
		if(cp < start): hi = mid - 1
		else if(cp > table[mid * 4 + 1]): lo = mid + 1
		else: {
			if(table[mid * 4 + 3] == 2 and (cp - start) % 2 != 0): func <- cp
			func <- cp + table[mid * 4 + 2]
		}
	}
	cp
}

// Returns the index of `cp` in a sorted key table, or `-1`.
const _special_index = fn(keys: [int], cp: int) int: {
	let lo = 0
	let hi = keys.len() - 1
	while(lo <= hi): {
		let mid = (lo + hi) / 2
		if(keys[mid] == cp): func <- mid
		if(keys[mid] < cp): lo = mid + 1 else: hi = mid - 1
	}
	-1
}

// Reports whether `cp` falls in a table of inclusive (start, end) pairs.
const _in_ranges = fn(table: [int], cp: int) bool: {
	let lo = 0
	let hi = table.len() / 2 - 1
	while(lo <= hi): {
		let mid = (lo + hi) / 2
		if(cp < table[mid * 2]): hi = mid - 1
		else if(cp > table[mid * 2 + 1]): lo = mid + 1
		else: func <- true
	}
	false
}

const _is_cased = fn(c: char) bool: _in_ranges(_CASED_RANGES, int(c))

const _is_case_ignorable = fn(c: char) bool: _in_ranges(_CASE_IGNORABLE_RANGES, int(c))

// The Final_Sigma condition (Unicode §3.13): the sigma at `i` follows a cased letter and is
// not followed by one, skipping case-ignorable characters (such as apostrophes and combining
// marks) in both directions.
const _is_final_sigma = fn(cs: [char], i: int) bool: {
	let j = i - 1
	while(j >= 0 and _is_case_ignorable(cs[j])): j -= 1
	if(j < 0 or not _is_cased(cs[j])): func <- false
	let k = i + 1
	while(k < cs.len() and _is_case_ignorable(cs[k])): k += 1
	not (k < cs.len() and _is_cased(cs[k]))
}

@test("unicode full case mapping, including length changes")
bind test_unicode_case_mapping = fn(): {
	assert(to_upper("straße") == "STRASSE", "ß uppercases to SS")
	assert(to_upper("ﬁne") == "FINE", "the fi ligature expands")
	assert(to_upper("ŉ") == "ʼN", "U+0149 expands to two characters")
	assert(to_lower("İstanbul") == "i̇stanbul", "dotted capital I lowercases to i + combining dot")
	assert(to_upper("ÀÉÎõü") == "ÀÉÎÕÜ" and to_lower("ÀÉÎÕÜ") == "àéîõü", "Latin-1 letters")
	assert(to_upper("ǆ") == "Ǆ" and to_lower("ǅ") == "ǆ", "digraph title case")
	assert(to_upper("привет мир") == "ПРИВЕТ МИР", "Cyrillic")
	assert(to_upper("ᏸ") == "Ᏸ", "Cherokee")
	assert(to_upper("𐐨") == "𐐀" and to_lower("𐐀") == "𐐨", "Deseret, outside the BMP")
	assert(to_upper("abc 123 !?") == "ABC 123 !?", "ASCII")
	assert(to_upper("日本語") == "日本語", "caseless scripts are unchanged")
	assert(to_upper("") == "", "empty")
}

@test("unicode Final_Sigma chooses ς at the end of a word")
bind test_unicode_final_sigma = fn(): {
	assert(to_lower("ΟΔΟΣ") == "οδος", "word-final sigma")
	assert(to_lower("ΣΟΦΙΑ") == "σοφια", "word-initial sigma")
	assert(to_lower("ΟΔΟΣ ΚΑΙ ΟΔΟΣ") == "οδος και οδος", "each word ends in ς")
	assert(to_lower("ΑΣΑ") == "ασα", "medial sigma")
	assert(to_lower("Σ") == "σ", "a lone sigma is not word-final")
	assert(to_lower("ΟΔΟΣ'") == "οδος'", "a case-ignorable apostrophe after it is skipped")
	assert(to_lower("ΟΔΟΣ.") == "οδος.", "punctuation ends the word")
	assert(fold("ΟΔΟΣ") == "οδοσ", "folding always uses σ")
}

@test("unicode case folding and case-insensitive comparison")
bind test_unicode_fold = fn(): {
	assert(fold("Straße") == "strasse", "ß folds to ss")
	assert(equal_fold("Straße", "STRASSE") and equal_fold("straße", "strasse"), "equal_fold")
	assert(equal_fold("ΟΔΟΣ", "οδος"), "final sigma folds like medial sigma")
	assert(not equal_fold("abc", "abd"), "different text")
	assert(fold("ẞ") == "ss", "capital sharp s")
	assert(UNICODE_VERSION.len() > 0, "version is recorded")
}
